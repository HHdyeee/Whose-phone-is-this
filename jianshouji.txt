六、编辑对话
查找 const story = [，其下方就是按先后顺序排列的消息。
每一行中，from 决定由谁说话，ko 是气泡中的台词，zh 是气泡下方的译文。

例如：
  {from:'me',ko:'你好。',zh:'안녕하세요.'},
这一行表示我方发送“你好。”，下方可展开韩文译文。
将 me 改为 them，这句话就会由对方发出。
me 对应右侧气泡，them 对应左侧气泡。正文和译文都可以使用自己需要的语言。

增加消息：复制一整行，粘贴到需要的位置，再修改说话人和内容。
删除消息：删除对应的整行。
调整顺序：将整行剪切到新的位置，页面会按从上到下的顺序播放。
同一人物可以连续发送多条消息；第一条写 them 时，对方会在进入聊天后先发言。

只显示正文时，将译文写为 zh:''，两个单引号之间留空。例如：
  {from:'me',ko:'你好。',zh:''},
需要在同一条消息中换行时，在换行位置写 \n。例如：
  {from:'me',ko:'第一行\n第二行',zh:''},

修改文字时，保留外侧的英文引号、逗号和大括号。
每条消息都可以沿用示例末尾的英文逗号，表示这一条填写结束。
文字中含有英文单引号时，可以改用英文双引号包住整段文字。例如：
  {from:'me',ko:"I'm here.",zh:'我在这里。'},

七、调整外观和播放速度
查找“页面样式”，可找到气泡颜色、背景颜色和字号等设置及其说明。
查找“常用设置”，可调整输入和回复的速度。
时间以毫秒填写：1000 毫秒等于 1 秒，500 毫秒等于半秒。
例如，把 replyDelayMs:380 改为 replyDelayMs:1000，
表示模拟已读后等待 1 秒，再显示对方的第一条回复。

八、阅读各部分代码
页面样式（CSS）：决定颜色、字号、大小和位置。
页面结构（HTML）：组成标题、按钮、列表、输入框等区域。
交互功能（JavaScript）：处理点击操作、逐字输入、回复和重播。
每个模块前都有中文注释，可以先阅读说明，再修改对应内容。
按钮和提示采用“中文（韩语）”的形式，韩文键盘上的字母按原样显示。


九、切换拼音键盘

  keyboardLayout:'korean',
将引号内的 korean 改为 pinyin，打开页面时就会使用拼音字母键盘：
  keyboardLayout:'pinyin',
korean 表示韩文键盘，pinyin 表示拼音键盘，两个名称均使用小写英文字母。

拼音键盘按 QWERTY 顺序排列，共三行字母键。
在页面中点击键盘左下方的 🌐，可以随时切换这两种布局。
拼音布局下，点击 ⇧ 可以切换字母大小写，再次点击恢复小写。

该键盘用于配合预设台词展示输入过程。中文台词填写在“对话内容”的 ko 后，
例如 ko:'你好，我们明天见。'，点击输入框后，这句话会逐字填入。
点击字母键可在输入框中追加字母；发送时显示的内容以 ko 中填写的台词为准。

      setReady(true);
      updateHint();
      return;

    activeDraft='';
    setReady(false);
    els.guide.textContent='正在自动输入消息…（메시지 자동 입력 중…）';

    const letters=Array.from(full);
    let i=0;

    typeTimer=setInterval(()=>{
      i=Math.min(i+2,letters.length);
      activeDraft=letters.slice(0,i).join('');
      els.input.value=activeDraft;

      if(i>=letters.length) {
        clearInterval(typeTimer);
        typeTimer=null;

        els.guide.textContent='可以发送了（전송할 준비가 됐어요）';
        setReady(true);
        updateHint();
      }
    },config.typingIntervalMs);

    scrollBottom();
  }

  // 十二、自动回复：依次显示连续的对方消息，遇到我方台词时暂停。

  async function replyFromContact() {
    const myRun=runId;

    busy=true;
    updateHint();

    await later(config.readDelayMs);
    if(myRun!==runId)return;

    markRead();

    // 等待常用设置中 replyDelayMs 指定的时长，然后继续显示回复。
    await later(config.replyDelayMs);

    if(myRun!==runId)return;

    while(index<story.length&&story[index].from==='them') {
      if(myRun!==runId)return;

      makeMessage(story[index]);
      index++;

      const delay=index>=story.length
        ?500
        :story[index].from==='them'
          ?config.replyGapMs+Math.floor(Math.random()*config.replyJitterMs)
          :480;

      await later(delay);
    }

    if(myRun!==runId)return;

    busy=false;
    updateHint();

    if(index>=story.length)toast(config.endText);
  }

  // 十三、发送消息：提交当前预设台词，并判断是否开始对方回复。

  function sendCurrent() {
    if(!composing||busy||index>=story.length||story[index].from!=='me')return;

    if(typeTimer) {
      fillDraft(true);
      return;
    }

    if(!els.input.value.trim())return;

    // 从“对话内容”中读取当前台词，并将它显示为一条已发送的消息。
    makeMessage(story[index]);
    index++;

    hideKeyboard();

    if(index<story.length&&story[index].from==='them') {
      replyFromContact();
    } else {
      updateHint();
      if(index>=story.length)toast(config.endText);
    }
  }

  els.send.addEventListener('click',()=>{
    if(!composing) {
      openKeyboard();
      return;
    }

    sendCurrent();
  });

  // 点击模拟键盘时，按键会短暂变色，并在输入框中追加或删除字符。
  // 点击发送后显示的台词以“对话内容”为准，可在上方 story 中修改实际台词。
  // 🌐 切换拼音和韩文布局；拼音布局中的 ⇧ 切换大小写；123 显示当前输入模式的提示。
  els.keyboard.addEventListener('click',e=>{
    const key=e.target.closest('[data-key]');
    if(!key)return;

    key.classList.add('pressed');
    setTimeout(()=>key.classList.remove('pressed'),110);

    const what=key.dataset.key;

    // 先处理布局按钮，让切换操作保留正在逐字填入的台词。
    if(what==='globe') {
      switchKeyboard();
      return;
    }
    if(what==='shift') {
      if(keyboardLayout==='pinyin') {
        uppercaseLetters=!uppercaseLetters;
        renderKeyboard();
      } else {
        toast('韩文键盘 · 自动输入模式（한글 키보드 · 자동 입력 모드）');
      }
      return;
    }
    if(what==='123') {
      toast(keyboardLayout==='pinyin'
        ?'拼音键盘 · 自动输入模式（병음 키보드 · 자동 입력 모드）'
        :'韩文键盘 · 自动输入模式（한글 키보드 · 자동 입력 모드）');
      return;
    }

    if(what==='send') {
      sendCurrent();
      return;
    }

    if(!composing)return;

    if(typeTimer)fillDraft(true);

    if(what==='delete') {
      els.input.value=Array.from(els.input.value).slice(0,-1).join('');
    } else if(what==='space') {
      els.input.value+=' ';

    } else {
      els.input.value+=what;
    }

    activeDraft=els.input.value;
    setReady(!!activeDraft.trim());
  });

  // 十四、搜索和辅助按钮：筛选联系人并显示操作说明。

  byId('searchOpen').addEventListener('click',()=>{
    const panel=byId('searchbox');
    panel.hidden=!panel.hidden;

    if(!panel.hidden)els.searchInput.focus();
  });

  els.searchInput.addEventListener('input',()=>{
    const matches=`${config.contactName} ${config.searchAliases}`.toLowerCase()
      .includes(els.searchInput.value.trim().toLowerCase());

    els.chatEntry.hidden=!matches;
  });

  byId('roomSearch').addEventListener('click',()=>{
    toast('可在各条消息的气泡下方查看译文（메시지 번역은 각 말풍선 아래에서 볼 수 있어요）');
  });

  byId('plusBtn').addEventListener('click',()=>{
    toast('可在右上角菜单中选择或下载头像（오른쪽 위 메뉴에서 프로필 사진을 선택하거나 다운로드할 수 있어요）');
  });

  // 十五、菜单操作：打开、关闭设置菜单，并响应 Escape 键。

  function openMenu() {
    els.overlay.hidden=false;
    byId('allTranslationsStatus').textContent=showAll?'开启（켜짐）':'关闭（꺼짐）';
  }

  byId('roomMenu').addEventListener('click',openMenu);
  byId('listSettings').addEventListener('click',openMenu);

  byId('sheetClose').addEventListener('click',()=>{
    els.overlay.hidden=true;
  });

  els.overlay.addEventListener('click',e=>{
    if(e.target===els.overlay)els.overlay.hidden=true;
  });

  document.addEventListener('keydown',e=>{
    if(e.key!=='Escape')return;

    if(!els.overlay.hidden) {
      els.overlay.hidden=true;
    } else if(composing) {
      hideKeyboard();
    } else if(enteredRoom) {
      showList();
    }
  });

  // 十六、重新播放：清空已显示消息，并恢复初始播放状态。

  function replay() {
    runId++;

    pendingUnreads.length=0;
    index=0;
    busy=false;

    hideKeyboard();

    els.scroll.querySelectorAll('.msg').forEach(m=>m.remove());

    els.entryPreview.textContent=config.initialPreview;
    els.entryTime.textContent='';

    // 重播时恢复常用设置中的默认译文显示状态。
    showAll=config.showTranslations;
    byId('allTranslationsStatus').textContent=showAll?'开启（켜짐）':'关闭（꺼짐）';

    // 保留配置中的故事日期，并刷新设备时钟。
    els.day.textContent=fictionalDate;
    renderClock();

    els.overlay.hidden=true;

    updateHint();
    scrollBottom();

    toast('正在从头重新开始（처음부터 다시 시작합니다）');
    if(enteredRoom && story.length && story[0].from==='them')replyFromContact();
  }

  byId('sheetReplay').addEventListener('click',replay);

  // 十七、批量切换译文：同步调整全部已显示译文及后续消息的默认状态。

  byId('sheetTranslations').addEventListener('click',()=>{
    showAll=!showAll;

    document.querySelectorAll('.translation').forEach(el=>{
      el.hidden=!showAll;
    });

    document.querySelectorAll('.translate-btn').forEach(el=>{
      el.setAttribute('aria-expanded',String(showAll));
    });

    byId('allTranslationsStatus').textContent=showAll?'开启（켜짐）':'关闭（꺼짐）';

    scrollBottom();
  });

  // 十八、头像文件操作：选择本地图片，或尝试下载当前头像。

  // 选择本地图片后，更新本次打开页面时的头像。要保存为模板的固定头像，请修改上方 avatarUrl。
  byId('sheetUpload').addEventListener('click',()=>byId('avatarUpload').click());
  byId('avatarUpload').addEventListener('change',event=>{
    const file=event.target.files[0];
    if(!file)return;
    if(!file.type.startsWith('image/')) {
      toast('请选择图片文件（이미지 파일을 선택해 주세요）');
      event.target.value='';
      return;
    }
    if(uploadedAvatarUrl)URL.revokeObjectURL(uploadedAvatarUrl);
    uploadedAvatarUrl=URL.createObjectURL(file);
    portrait=uploadedAvatarUrl;
    refreshAvatars();
    els.overlay.hidden=true;
    event.target.value='';
    toast('头像仅在当前页面有效（사진은 현재 페이지에서만 적용돼요）');
  });

  // 下载按钮会读取当前头像，并将图片交给浏览器保存。
  // 图片能否直接下载取决于浏览器和图片所在网站的设置，也可打开原图片网址保存。
  byId('sheetPhoto').addEventListener('click',async()=>{
    els.overlay.hidden=true;

    if(!portrait) {
      toast('请先设置头像（먼저 프로필 사진을 설정해 주세요）');
      return;
    }

    try {
      const response=await fetch(portrait);

      if(!response.ok) {
        throw new Error('Image request failed');
      }

      const blob=await response.blob();
      const url=URL.createObjectURL(blob);

      const link=document.createElement('a');
      link.href=url;
      link.download='avatar'; // 下载文件使用 avatar 作为名称，图片格式由浏览器处理。

      document.body.appendChild(link);
      link.click();
      link.remove();

      setTimeout(()=>{
        URL.revokeObjectURL(url);
      },2000);

    } catch(error) {
      toast('无法下载头像（프로필 사진을 다운로드할 수 없어요）');
    }
  });

  // 初始化提示；对话将在进入聊天页面并按相应操作后开始。
  updateHint();
})();
</script>
